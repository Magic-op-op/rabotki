# rabotki
a = float(input("Введите первое число: "))
b = float(input("Введите второе число: "))
print(f"Сумма: {a + b:.2f}")
print(f"Разность: {a - b:.2f}")
print(f"Произведение: {a * b:.2f}")

if b != 0:
    print(f"Частное: {a / b:.2f}")
    print(f"Целая часть от деления: {a // b:.1f}")
    print(f"Остаток от деления: {a % b}")
else:
    print("Частное: деление на ноль невозможно")
    print("Целая часть от деления: деление на ноль невозможно")
    print("Остаток от деления: деление на ноль невозможно")

print(f"Возведение в степень: {a ** b:.2f}")
